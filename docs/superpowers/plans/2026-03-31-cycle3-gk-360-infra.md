```python
# src/analytics/goalkeeper.py
"""Goalkeeper analytics — distribution xT, collection stats, PSxG, sweeper metrics.

Four-pillar GK evaluation taxonomy:
  1. Shot stopping (D39: PSxG)
  2. Distribution (D38: xT delta on GK passes)
  3. Cross collection (D38: claim/punch rates)
  4. Defensive activity (D39: sweeper positioning)

References:
  - Butcher et al. (2025), "An Expected Goals On Target (xGOT) Model" (MDPI)
  - Lamberts (2025), Goalkeeper Value Model
  - Yam, "A Data-Driven GK Evaluation Framework" (MIT Sloan)
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
import pandas as pd


@dataclass(frozen=True)
class GKDistributionResult:
    """Per player-match GK distribution summary."""

    total_xt_added: float
    xt_per_pass: float
    pass_count: int
    short_pct: float
    medium_pct: float
    long_pct: float
    launch_rate: float


# SPADL pitch dimensions (105 x 68 metres).
_PITCH_LENGTH = 105.0
_PITCH_WIDTH = 68.0
_N_ZONES_X = 12
_N_ZONES_Y = 8

# Pass length thresholds (metres).
_SHORT_THRESHOLD = 32.0
_LONG_THRESHOLD = 60.0


def _xt_zone_value(x: float, y: float, xt_grid: np.ndarray) -> float:
    """Look up xT value for a (x, y) coordinate on the SPADL 105x68 pitch."""
    zone_x = max(0, min(int(x / (_PITCH_LENGTH / _N_ZONES_X)), _N_ZONES_X - 1))
    zone_y = max(0, min(int(y / (_PITCH_WIDTH / _N_ZONES_Y)), _N_ZONES_Y - 1))
    return float(xt_grid[zone_x, zone_y])


def compute_gk_distribution_xt(
    passes_df: pd.DataFrame,
    xt_grid: np.ndarray,
) -> pd.DataFrame:
    """Compute xT delta for GK-initiated passes.

    Args:
        passes_df: GK passes with columns ``player_id, match_id, start_x,
            start_y, end_x, end_y, action_result``. SPADL 105x68 coordinates.
        xt_grid: Expected Threat grid, shape ``(12, 8)``.

    Returns:
        DataFrame with one row per ``(player_id, match_id)``. Columns:
        ``player_id, match_id, total_xt_added, xt_per_pass, pass_count,
        short_pct, medium_pct, long_pct, launch_rate``.
    """
    if passes_df.empty:
        return pd.DataFrame(
            columns=[
                "player_id", "match_id", "total_xt_added", "xt_per_pass",
                "pass_count", "short_pct", "medium_pct", "long_pct", "launch_rate",
            ]
        )

    # Vectorised xT lookup per pass.
    start_xt = np.array([
        _xt_zone_value(row["start_x"], row["start_y"], xt_grid)
        for _, row in passes_df.iterrows()
    ])
    end_xt = np.array([
        _xt_zone_value(row["end_x"], row["end_y"], xt_grid)
        for _, row in passes_df.iterrows()
    ])
    df = passes_df.copy()
    df["xt_delta"] = end_xt - start_xt

    # Pass length classification.
    dx = df["end_x"].to_numpy() - df["start_x"].to_numpy()
    dy = df["end_y"].to_numpy() - df["start_y"].to_numpy()
    df["distance"] = np.sqrt(dx**2 + dy**2)
    df["length_class"] = np.where(
        df["distance"] < _SHORT_THRESHOLD, "short",
        np.where(df["distance"] >= _LONG_THRESHOLD, "long", "medium"),
    )

    # Aggregate per player-match.
    rows: list[dict] = []
    for (pid, mid), group in df.groupby(["player_id", "match_id"]):
        n = len(group)
        total_xt = float(group["xt_delta"].sum())
        length_counts = group["length_class"].value_counts()
        rows.append({
            "player_id": pid,
            "match_id": mid,
            "total_xt_added": total_xt,
            "xt_per_pass": total_xt / n if n else 0.0,
            "pass_count": n,
            "short_pct": int(length_counts.get("short", 0)) / n if n else 0.0,
            "medium_pct": int(length_counts.get("medium", 0)) / n if n else 0.0,
            "long_pct": int(length_counts.get("long", 0)) / n if n else 0.0,
            "launch_rate": int(length_counts.get("long", 0)) / n if n else 0.0,
        })
    return pd.DataFrame(rows)
```

```python
def compute_gk_collection_stats(actions_df: pd.DataFrame) -> pd.DataFrame:
    """Compute GK claim/punch collection statistics.

    Args:
        actions_df: All actions for GK players. Filtered internally to
            ``keeper_claim`` and ``keeper_punch``.

    Returns:
        DataFrame with one row per ``(player_id, match_id)``. Columns:
        ``player_id, match_id, claims, claim_success_rate, punches``.
    """
    if actions_df.empty:
        return pd.DataFrame(columns=["player_id", "match_id", "claims", "claim_success_rate", "punches"])

    collection = actions_df[actions_df["action_type"].isin({"keeper_claim", "keeper_punch"})].copy()
    if collection.empty:
        return pd.DataFrame(columns=["player_id", "match_id", "claims", "claim_success_rate", "punches"])

    rows: list[dict] = []
    for (pid, mid), group in collection.groupby(["player_id", "match_id"]):
        claims = group[group["action_type"] == "keeper_claim"]
        punches = group[group["action_type"] == "keeper_punch"]
        n_claims = len(claims)
        n_success = int((claims["action_result"] == "success").sum()) if n_claims else 0
        rows.append({
            "player_id": pid,
            "match_id": mid,
            "claims": n_claims,
            "claim_success_rate": n_success / n_claims if n_claims else 0.0,
            "punches": len(punches),
        })
    return pd.DataFrame(rows)


def compute_gk_action_summary(
    passes_df: pd.DataFrame,
    actions_df: pd.DataFrame,
    xt_grid: np.ndarray,
) -> pd.DataFrame:
    """Combine distribution xT, collection stats, and save/pick-up counts.

    Args:
        passes_df: GK passes (same schema as ``compute_gk_distribution_xt``).
        actions_df: All GK actions (must include ``action_type`` column).
        xt_grid: Expected Threat grid, shape ``(12, 8)``.

    Returns:
        DataFrame with one row per ``(player_id, match_id)``. Full column set
        for ``fct_goalkeeper_stats``.
    """
    dist = compute_gk_distribution_xt(passes_df, xt_grid)
    coll = compute_gk_collection_stats(actions_df)

    # Count saves and pick-ups directly.
    save_counts: dict[tuple, dict] = {}
    for (pid, mid), group in actions_df.groupby(["player_id", "match_id"]):
        save_counts[(pid, mid)] = {
            "saves": int((group["action_type"] == "keeper_save").sum()),
            "save_pct": (
                int(((group["action_type"] == "keeper_save") & (group["action_result"] == "success")).sum())
                / max(int((group["action_type"] == "keeper_save").sum()), 1)
            ),
            "keeper_pick_ups": int((group["action_type"] == "keeper_pick_up").sum()),
        }

    # Merge all sources.
    keys = set()
    for df in [dist, coll]:
        if not df.empty:
            keys.update(zip(df["player_id"], df["match_id"]))
    keys.update(save_counts.keys())

    rows: list[dict] = []
    for pid, mid in keys:
        row: dict = {"player_id": pid, "match_id": mid}
        # Distribution.
        d = dist[(dist["player_id"] == pid) & (dist["match_id"] == mid)]
        if not d.empty:
            for col in ["total_xt_added", "xt_per_pass", "pass_count", "short_pct", "medium_pct", "long_pct", "launch_rate"]:
                row[col] = d.iloc[0][col]
        else:
            for col in ["total_xt_added", "xt_per_pass", "pass_count", "short_pct", "medium_pct", "long_pct", "launch_rate"]:
                row[col] = 0.0 if col != "pass_count" else 0
        # Collection.
        c = coll[(coll["player_id"] == pid) & (coll["match_id"] == mid)]
        if not c.empty:
            for col in ["claims", "claim_success_rate", "punches"]:
                row[col] = c.iloc[0][col]
        else:
            row.update({"claims": 0, "claim_success_rate": 0.0, "punches": 0})
        # Saves.
        sc = save_counts.get((pid, mid), {"saves": 0, "save_pct": 0.0, "keeper_pick_ups": 0})
        row.update(sc)
        rows.append(row)

    return pd.DataFrame(rows)
```

```python
def _make_on_target_shots(n: int = 100) -> pd.DataFrame:
    """Synthetic on-target shots with known goalmouth coordinates.

    Goals clustered at corners (high z, extreme y).
    Saves clustered at center (low z, mid y).
    """
    rng = np.random.default_rng(42)
    goals = n // 2
    saves = n - goals
    # Goals: high corners.
    goal_y = rng.uniform(36.0, 37.0, size=goals)  # left side of goal
    goal_z = rng.uniform(6.0, 8.0, size=goals)     # top
    # Saves: centre of goal.
    save_y = rng.uniform(39.0, 41.0, size=saves)   # centre
    save_z = rng.uniform(0.5, 2.0, size=saves)      # low
    return pd.DataFrame({
        "event_id": [f"e{i}" for i in range(n)],
        "match_id": ["m1"] * n,
        "player_id": [10] * n,  # shooter, not GK
        "end_location_y": np.concatenate([goal_y, save_y]),
        "end_location_z": np.concatenate([goal_z, save_z]),
        "shot_outcome": ["Goal"] * goals + ["Saved"] * saves,
        "is_goal": [1] * goals + [0] * saves,
    })
```

```python
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler


@dataclass(frozen=True)
class PSxGModel:
    """Trained Post-Shot Expected Goals model.

    Logistic regression on goalmouth coordinates following
    Butcher et al. (2025).
    """

    coefficients: np.ndarray
    intercept: float
    scaler_mean: np.ndarray
    scaler_scale: np.ndarray


# StatsBomb goal frame: y ∈ [36, 44] (8-yard goal centred at y=40), z ∈ [0, 8].
_GOAL_Y_MIN = 36.0
_GOAL_Y_MAX = 44.0
_GOAL_Z_MAX = 8.0


def _normalise_goalmouth(y: np.ndarray, z: np.ndarray) -> np.ndarray:
    """Normalise StatsBomb goalmouth coords to [0, 1] range."""
    y_norm = (y - _GOAL_Y_MIN) / (_GOAL_Y_MAX - _GOAL_Y_MIN)
    z_norm = z / _GOAL_Z_MAX
    return np.column_stack([y_norm, z_norm])


def train_psxg_model(shots_df: pd.DataFrame) -> PSxGModel:
    """Train a PSxG logistic regression on on-target shots.

    Args:
        shots_df: On-target shots with columns ``end_location_y``,
            ``end_location_z``, ``is_goal`` (1/0).

    Returns:
        Frozen ``PSxGModel`` with fitted parameters.
    """
    on_target = shots_df.dropna(subset=["end_location_z"]).copy()
    features = _normalise_goalmouth(
        on_target["end_location_y"].to_numpy(),
        on_target["end_location_z"].to_numpy(),
    )
    scaler = StandardScaler()
    X = scaler.fit_transform(features)
    y = on_target["is_goal"].to_numpy().astype(int)

    lr = LogisticRegression(max_iter=1000, random_state=42)
    lr.fit(X, y)

    return PSxGModel(
        coefficients=lr.coef_[0],
        intercept=float(lr.intercept_[0]),
        scaler_mean=scaler.mean_,
        scaler_scale=scaler.scale_,
    )


def predict_psxg(model: PSxGModel, shots_df: pd.DataFrame) -> pd.DataFrame:
    """Predict PSxG for each shot. Off-target shots get ``psxg = NaN``.

    Args:
        model: Trained ``PSxGModel``.
        shots_df: Shots with columns ``end_location_y``, ``end_location_z``.

    Returns:
        Input DataFrame with ``psxg`` column added.
    """
    result = shots_df.copy()
    result["psxg"] = np.nan
    on_target_mask = result["end_location_z"].notna()
    if not on_target_mask.any():
        return result

    on_target = result.loc[on_target_mask]
    features = _normalise_goalmouth(
        on_target["end_location_y"].to_numpy(),
        on_target["end_location_z"].to_numpy(),
    )
    X = (features - model.scaler_mean) / model.scaler_scale

    # Manual logistic prediction (no sklearn dependency at inference).
    logits = X @ model.coefficients + model.intercept
    probs = 1.0 / (1.0 + np.exp(-logits))
    result.loc[on_target_mask, "psxg"] = probs
    return result


def compute_goals_prevented(gk_df: pd.DataFrame) -> pd.DataFrame:
    """Compute goals prevented = PSxG faced - goals conceded.

    Args:
        gk_df: DataFrame with ``player_id, match_id, psxg_faced, goals_conceded``.

    Returns:
        Input DataFrame with ``goals_prevented`` column added.
    """
    result = gk_df.copy()
    result["goals_prevented"] = result["psxg_faced"] - result["goals_conceded"]
    return result
```

```python
# SPADL penalty area x-extent (metres from own goal line).
_PENALTY_AREA_X = 16.5


def compute_sweeper_metrics(events_df: pd.DataFrame) -> pd.DataFrame:
    """Compute sweeper-keeper positioning metrics.

    Args:
        events_df: GK events with columns ``player_id, match_id, start_x,
            start_y, action_type, minutes_played``. SPADL coordinates
            (own goal at x=0).

    Returns:
        DataFrame with one row per ``(player_id, match_id)``. Columns:
        ``player_id, match_id, avg_defensive_action_distance,
        actions_outside_box_per_90``.
    """
    if events_df.empty:
        return pd.DataFrame(
            columns=["player_id", "match_id", "avg_defensive_action_distance", "actions_outside_box_per_90"]
        )

    rows: list[dict] = []
    for (pid, mid), group in events_df.groupby(["player_id", "match_id"]):
        avg_dist = float(group["start_x"].mean())
        outside_box = int((group["start_x"] > _PENALTY_AREA_X).sum())
        minutes = float(group["minutes_played"].iloc[0])
        per_90 = outside_box * 90.0 / max(minutes, 1.0)
        rows.append({
            "player_id": pid,
            "match_id": mid,
            "avg_defensive_action_distance": avg_dist,
            "actions_outside_box_per_90": per_90,
        })
    return pd.DataFrame(rows)
```

```python
def load_shots() -> pd.DataFrame:
    """Download on-target shots from HF Hub."""
    path = hf_hub_download(
        repo_id=INPUT_DATASET,
        filename="data/shots_on_target.parquet",
        repo_type="dataset",
    )
    df = pd.read_parquet(path)
    logger.info("Loaded %d on-target shots", len(df))
    return df


def evaluate_model(model: PSxGModel, test_df: pd.DataFrame) -> dict:
    """Evaluate PSxG model on test set."""
    result = predict_psxg(model, test_df)
    y_true = result["is_goal"].to_numpy()
    y_pred = result["psxg"].to_numpy()
    metrics = {
        "log_loss": float(log_loss(y_true, y_pred)),
        "brier_score": float(brier_score_loss(y_true, y_pred)),
        "n_test": len(test_df),
        "goal_rate_test": float(y_true.mean()),
        "mean_psxg_test": float(y_pred.mean()),
    }
    logger.info("Evaluation: %s", json.dumps(metrics, indent=2))
    return metrics


def publish_model(model: PSxGModel, metrics: dict) -> None:
    """Publish PSxG model weights and config to HF Hub."""
    api = HfApi()
    api.create_repo(OUTPUT_MODEL, exist_ok=True)

    with tempfile.TemporaryDirectory() as tmpdir:
        model_path = Path(tmpdir) / "psxg_model.json"
        model_data = {
            "coefficients": model.coefficients.tolist(),
            "intercept": model.intercept,
            "scaler_mean": model.scaler_mean.tolist(),
            "scaler_scale": model.scaler_scale.tolist(),
            "metrics": metrics,
        }
        model_path.write_text(json.dumps(model_data, indent=2))
        api.upload_file(
            path_or_fileobj=str(model_path),
            path_in_repo="psxg_model.json",
            repo_id=OUTPUT_MODEL,
        )
    logger.info("Published model to %s", OUTPUT_MODEL)


def publish_predictions(shots_df: pd.DataFrame, model: PSxGModel) -> None:
    """Generate and publish PSxG predictions for all shots."""
    result = predict_psxg(model, shots_df)
    output_cols = ["event_id", "match_id", "player_id", "psxg"]
    out = result[output_cols]

    api = HfApi()
    api.create_repo(OUTPUT_PREDICTIONS, repo_type="dataset", exist_ok=True)

    with tempfile.TemporaryDirectory() as tmpdir:
        parquet_path = Path(tmpdir) / "psxg_predictions.parquet"
        out.to_parquet(parquet_path, index=False)
        api.upload_file(
            path_or_fileobj=str(parquet_path),
            path_in_repo="data/psxg_predictions.parquet",
            repo_id=OUTPUT_PREDICTIONS,
            repo_type="dataset",
        )
    logger.info("Published %d predictions to %s", len(out), OUTPUT_PREDICTIONS)


def main() -> None:
    """Train, evaluate, and publish PSxG model."""
    mlflow.set_experiment("psxg-training")

    shots = load_shots()
    train_df, test_df = train_test_split(shots, test_size=0.2, random_state=42, stratify=shots["is_goal"])

    with mlflow.start_run(run_name="psxg-logistic"):
        model = train_psxg_model(train_df)
        metrics = evaluate_model(model, test_df)

        mlflow.log_params({"model_type": "logistic_regression", "features": "end_location_y,end_location_z"})
        mlflow.log_metrics(metrics)

        publish_model(model, metrics)
        publish_predictions(shots, model)

    logger.info("PSxG training complete")


if __name__ == "__main__":
    main()
```

```markdown
Logistic regression model predicting goal probability from goalmouth shot placement, following Butcher et al. (2025). Trained on ~15K StatsBomb on-target shots.

## Features

Two features from StatsBomb `shot.end_location`:

import json, numpy as np
from huggingface_hub import hf_hub_download

path = hf_hub_download("luxury-lakehouse/psxg-model", "psxg_model.json")

coef = np.array(model["coefficients"])
intercept = model["intercept"]
mean, scale = np.array(model["scaler_mean"]), np.array(model["scaler_scale"])

features = np.array([[0.5, 0.9]])  # centre-top of goal
X = (features - mean) / scale
logits = X @ coef + intercept
psxg = 1.0 / (1.0 + np.exp(-logits))
```

```python
# src/analytics/football2vec_360.py
"""Football2Vec 360-enriched encoder — transformer + Deep Sets → 144d.

Extends the Football2Vec v2 transformer encoder with a Deep Sets branch
that encodes StatsBomb 360 freeze frame context per action. Output is
128d (transformer) + 16d (Deep Sets) = 144d.

References:
  - Theiner et al. (2022), football2vec
  - Zaheer et al. (2017), Deep Sets (NeurIPS)
  - Danesi (2025), adversarial team debiasing
"""

from __future__ import annotations

from dataclasses import dataclass

import torch
import torch.nn as nn

from analytics.football2vec_transformer import Football2VecConfig


@dataclass(frozen=True)
class Football2Vec360Config(Football2VecConfig):
    """Config for the 360-enriched encoder."""

    context_dim: int = 16
    deep_sets_hidden: int = 32
    player_feature_dim: int = 4  # [x_norm, y_norm, is_keeper, is_teammate]
    use_pretrained_encoder: bool = True


class _DeepSetsEncoder(nn.Module):
    """Permutation-invariant Deep Sets encoder for freeze frame players.

    Architecture: Linear(4→32) → ReLU → Linear(32→16) → ReLU → sum.
    Matches the NumPy implementation in ``set_encoder.py``.
    """

    def __init__(self, cfg: Football2Vec360Config) -> None:
        super().__init__()
        self.phi = nn.Sequential(
            nn.Linear(cfg.player_feature_dim, cfg.deep_sets_hidden),
            nn.ReLU(),
            nn.Linear(cfg.deep_sets_hidden, cfg.context_dim),
            nn.ReLU(),
        )

    def forward(self, player_features: torch.Tensor) -> torch.Tensor:
        """Encode a set of player features.

        Args:
            player_features: ``(batch, seq_len, max_players, 4)`` or
                ``(batch, seq_len, context_dim)`` if pre-encoded.

        Returns:
            ``(batch, seq_len, context_dim)`` — sum-pooled context vectors.
        """
        if player_features.dim() == 4:
            # Raw player features — encode and sum-pool.
            h = self.phi(player_features)  # (batch, seq, players, context_dim)
            return h.sum(dim=2)  # (batch, seq, context_dim)
        # Already pre-encoded (context_dim per action).
        return player_features


class Football2Vec360Encoder(nn.Module):
    """Football2Vec v2 transformer + Deep Sets 360 context → 144d output."""

    def __init__(self, cfg: Football2Vec360Config) -> None:
        super().__init__()
        self.cfg = cfg

        # Transformer encoder components (same as Football2VecEncoder).
        self.token_embedding = nn.Embedding(cfg.vocab_size, cfg.hidden_dim)
        self.spatial_x = nn.Sequential(
            nn.Linear(1, cfg.spatial_mlp_dim), nn.ReLU(),
            nn.Linear(cfg.spatial_mlp_dim, cfg.hidden_dim),
        )
        self.spatial_y = nn.Sequential(
            nn.Linear(1, cfg.spatial_mlp_dim), nn.ReLU(),
            nn.Linear(cfg.spatial_mlp_dim, cfg.hidden_dim),
        )
        self.position_embedding = nn.Embedding(cfg.max_seq_len, cfg.hidden_dim)
        self.embedding_dropout = nn.Dropout(cfg.dropout)

        encoder_layer = nn.TransformerEncoderLayer(
            d_model=cfg.hidden_dim,
            nhead=cfg.num_heads,
            dim_feedforward=cfg.hidden_dim * 4,
            dropout=cfg.dropout,
            batch_first=True,
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=cfg.num_layers)

        # Deep Sets 360 context branch.
        self.deep_sets = _DeepSetsEncoder(cfg)

    def forward(
        self,
        action_ids: torch.Tensor,
        x_coords: torch.Tensor,
        y_coords: torch.Tensor,
        attention_mask: torch.Tensor,
        context_360: torch.Tensor | None = None,
    ) -> torch.Tensor:
        """Forward pass producing 144d embeddings.

        Args:
            action_ids: ``(batch, seq_len)`` — SPADL action type IDs.
            x_coords: ``(batch, seq_len)`` — normalised x coordinates.
            y_coords: ``(batch, seq_len)`` — normalised y coordinates.
            attention_mask: ``(batch, seq_len)`` — True for real tokens.
            context_360: ``(batch, seq_len, context_dim)`` or
                ``(batch, seq_len, max_players, 4)``. None → zeros.

        Returns:
            ``(batch, 144)`` — mean-pooled transformer (128d) + Deep Sets (16d).
        """
        batch, seq_len = action_ids.shape

        # Transformer encoding.
        tok = self.token_embedding(action_ids)
        sx = self.spatial_x(x_coords.unsqueeze(-1))
        sy = self.spatial_y(y_coords.unsqueeze(-1))
        pos = self.position_embedding(torch.arange(seq_len, device=action_ids.device).unsqueeze(0).expand(batch, -1))
        embedded = self.embedding_dropout(tok + sx + sy + pos)

        # Transformer forward with padding mask.
        padding_mask = ~attention_mask
        encoded = self.transformer(embedded, src_key_padding_mask=padding_mask)

        # Mean pooling over valid tokens.
        mask_expanded = attention_mask.unsqueeze(-1).float()
        summed = (encoded * mask_expanded).sum(dim=1)
        lengths = mask_expanded.sum(dim=1).clamp(min=1)
        transformer_out = summed / lengths  # (batch, 128)

        # Deep Sets 360 context.
        if context_360 is None:
            context_out = torch.zeros(batch, self.cfg.context_dim, device=action_ids.device)
        else:
            ds_out = self.deep_sets(context_360)  # (batch, seq_len, context_dim)
            ds_masked = (ds_out * mask_expanded).sum(dim=1)
            context_out = ds_masked / lengths  # (batch, context_dim)

        # Concatenate → 144d.
        return torch.cat([transformer_out, context_out], dim=-1)
```

```python
"""Import space creation values from HF Hub to bronze Delta table.

Usage (Databricks):
    import_space_creation --catalog soccer_analytics --schema bronze \
        --volume-path /Volumes/soccer_analytics/dev_gold/model_weights/space_creation
"""

from __future__ import annotations

import argparse
import logging
import re
import sys

from huggingface_hub import hf_hub_download
from pyspark.sql import SparkSession
import pyspark.sql.functions as spark_fn

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
logger = logging.getLogger(__name__)

_IDENTIFIER_RE = re.compile(r"^[a-zA-Z_][a-zA-Z0-9_]*$")
HF_REPO = "luxury-lakehouse/space-creation-values"
TABLE_NAME = "space_creation_values"


def main() -> None:
    parser = argparse.ArgumentParser(description="Import space creation values from HF Hub")
    parser.add_argument("--catalog", default="soccer_analytics")
    parser.add_argument("--schema", default="bronze")
    parser.add_argument("--volume-path", required=True, help="UC Volume staging path")
    args = parser.parse_args()

    for name, val in [("catalog", args.catalog), ("schema", args.schema)]:
        if not _IDENTIFIER_RE.match(val):
            logger.error("Invalid %s identifier: %s", name, val)
            sys.exit(1)

    # Step 1: Download from HF Hub to local cache.
    logger.info("Downloading from %s", HF_REPO)
    local_path = hf_hub_download(
        repo_id=HF_REPO,
        filename="data/space_creation_values.parquet",
        repo_type="dataset",
    )

    # Step 2: Copy to UC Volume.
    spark = SparkSession.builder.getOrCreate()
    volume_file = f"{args.volume_path}/space_creation_values.parquet"
    logger.info("Copying to UC Volume: %s", volume_file)
    dbutils = spark._jvm.com.databricks.dbutils_v1.DBUtilsHolder.dbutils()  # type: ignore[union-attr]
    dbutils.fs.cp(f"file:{local_path}", volume_file)

    # Step 3: Read from Volume and write to Delta.
    df = spark.read.parquet(volume_file)
    df = df.withColumn("_ingested_at", spark_fn.current_timestamp())
    row_count = df.count()
    logger.info("Read %d rows from Volume", row_count)

    if row_count == 0:
        logger.warning("No rows found — skipping write")
        return

    table = f"{args.catalog}.{args.schema}.{TABLE_NAME}"
    match_ids = [str(r["match_id"]) for r in df.select("match_id").distinct().collect()]
    quoted = ", ".join(f"'{mid}'" for mid in match_ids)
    predicate = f"match_id IN ({quoted})"

    logger.info("Writing %d rows to %s (replaceWhere: %s)", row_count, table, predicate)
    df.write.format("delta") \
        .option("mergeSchema", "true") \
        .option("replaceWhere", predicate) \
        .mode("overwrite") \
        .saveAsTable(table)

    logger.info("Import complete: %d rows → %s", row_count, table)


if __name__ == "__main__":
    main()
```

## Task 15: Synced Table Maintenance Wrapper (TD#2)

**Files:**
- Create: `scripts/maintain_synced_tables.py`

- [ ] **Step 1: Create the wrapper script**

```python
# scripts/maintain_synced_tables.py
"""Synced table maintenance: refresh → create indexes → verify.

Orchestrates the operational procedure for synced table maintenance
as a single command, encoding what was previously a manual sequence.

Usage:
    python scripts/maintain_synced_tables.py --catalog soccer_analytics --schema dev_gold
    python scripts/maintain_synced_tables.py --skip-refresh   # indexes only
    python scripts/maintain_synced_tables.py --dry-run         # print commands
"""

from __future__ import annotations

import argparse
import json
import logging
import subprocess
import sys
import time

logging.basicConfig(
    level=logging.INFO,
    format='{"time":"%(asctime)s","level":"%(levelname)s","message":"%(message)s"}',
)
logger = logging.getLogger(__name__)


def _run_step(name: str, cmd: list[str], *, dry_run: bool = False) -> int:
    """Run a subprocess step with structured logging."""
    logger.info("Step %s: %s", name, " ".join(cmd))
    if dry_run:
        logger.info("[DRY RUN] Would execute: %s", " ".join(cmd))
        return 0

    start = time.monotonic()
    result = subprocess.run(cmd, capture_output=False)
    elapsed = time.monotonic() - start
    logger.info(
        "Step %s completed in %.1fs with exit code %d",
        name, elapsed, result.returncode,
    )
    return result.returncode


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Synced table maintenance: refresh → create indexes → verify",
    )
    parser.add_argument("--catalog", default="soccer_analytics")
    parser.add_argument("--schema", default="dev_gold")
    parser.add_argument("--dry-run", action="store_true", help="Print commands without executing")
    parser.add_argument("--skip-refresh", action="store_true", help="Skip synced table refresh (indexes only)")
    parser.add_argument("--skip-verify", action="store_true", help="Skip index verification")
    args = parser.parse_args()

    python = sys.executable
    scripts_dir = "scripts"

    # Step 1: Refresh synced tables.
    if not args.skip_refresh:
        rc = _run_step(
            "refresh",
            [python, f"{scripts_dir}/refresh_synced_tables.py", "--wait",
             "--catalog", args.catalog, "--schema", args.schema],
            dry_run=args.dry_run,
        )
        if rc != 0:
            logger.error("Refresh failed (exit %d) — aborting", rc)
            sys.exit(rc)

    # Step 2: Create indexes.
    rc = _run_step(
        "create-indexes",
        [python, f"{scripts_dir}/create_indexes.py",
         "--catalog", args.catalog, "--schema", args.schema],
        dry_run=args.dry_run,
    )
    if rc != 0:
        logger.error("Index creation failed (exit %d) — aborting", rc)
        sys.exit(rc)

    # Step 3: Verify indexes.
    if not args.skip_verify:
        rc = _run_step(
            "verify-indexes",
            [python, f"{scripts_dir}/create_indexes.py", "--verify",
             "--catalog", args.catalog, "--schema", args.schema],
            dry_run=args.dry_run,
        )
        if rc != 0:
            logger.warning("Verification reported issues (exit %d) — see warnings above", rc)
            # Soft exit — verify warnings are not fatal.

    logger.info("Maintenance complete")


if __name__ == "__main__":
    main()
```
